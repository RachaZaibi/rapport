\begin{itemize}
    \item \textbf{Caméras pour les machines plus anciennes} :
    \begin{itemize}
        \item \textbf{Type} : Caméras USB haute résolution (ex. Logitech C920).
        \item \textbf{Rôle} : Capture des écrans des machines anciennes pour l'extraction des données via OCR (Reconnaissance Optique de Caractères).
        \item \textbf{Configuration} : Montage fixe sur les moniteurs pour une capture stable et précise.
    \end{itemize}

    \item \textbf{Serveurs pour le stockage et le traitement des données} :
    \begin{itemize}
        \item \textbf{Type} : Serveurs dédiés ou cloud (ex. AWS, Azure, ou serveurs locaux).
        \item \textbf{Rôle} : Stockage des données collectées, traitement en temps réel, et génération des alertes.
        \item \textbf{Spécifications techniques} :
        \begin{itemize}
            \item \textbf{Processeur} : Multi-core (ex. Intel Xeon ou AMD EPYC).
            \item \textbf{Mémoire RAM} : 16 Go minimum (32 Go recommandé pour une meilleure performance).
            \item \textbf{Stockage} : Disque SSD de 500 Go minimum pour un accès rapide aux données.
            \item \textbf{Connectivité} : Réseau haut débit (1 Gbps) pour une transmission rapide des données.
        \end{itemize}
    \end{itemize}

    \item \textbf{Réseau de communication} :
    \begin{itemize}
        \item \textbf{Type} : Réseau local (LAN) sécurisé avec accès à Internet.
        \item \textbf{Rôle} : Assurer la communication entre les caméras, les serveurs et les applications mobiles/web.
        \item \textbf{Exigences} :
        \begin{itemize}
            \item \textbf{Bande passante} : 100 Mbps minimum pour une transmission fluide des données.
            \item \textbf{Sécurité} : Chiffrement des données (ex. SSL/TLS) pour protéger les informations médicales sensibles.
        \end{itemize}
    \end{itemize}

    \item \textbf{Appareils mobiles pour le personnel médical} :
    \begin{itemize}
        \item \textbf{Type} : Smartphones et tablettes compatibles Android et iOS.
        \item \textbf{Rôle} : Recevoir des notifications push en temps réel et accéder à l'interface mobile du système.
        \item \textbf{Exigences} :
        \begin{itemize}
            \item \textbf{Version minimale} : Android 8.0 (Oreo) ou iOS 12.
            \item \textbf{Connexion} : Wi-Fi ou 4G/5G pour une réception rapide des alertes.
        \end{itemize}
    \end{itemize}
\end{itemize}
